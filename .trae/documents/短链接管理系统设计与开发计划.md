```prisma
model User {

  id            String        @id @default(cuid())

  referrer      String?

}
```

### 2. 短链接生成逻辑

```typescript
// src/lib/utils/shortenUrl.ts
const generateSlug = (length: number = 6): string => {
  const chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  let slug = '';
  for (let i = 0; i < length; i++) {
    slug += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return slug;
};

export const createUniqueSlug = async (prisma: PrismaClient, customSlug?: string): Promise<string> => {
  if (customSlug) {
    const existing = await prisma.link.findUnique({ where: { slug: customSlug } });
    if (!existing) return customSlug;
    throw new Error('Custom slug already exists');
  }
  
  let slug: string;
  let existing: Link | null;
  do {
    slug = generateSlug();
    existing = await prisma.link.findUnique({ where: { slug } });
  } while (existing);
  
  return slug;
};
```

### 3. 重定向逻辑

```typescript
// functions/redirect.ts
import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

export default async (req: Request) => {
  const url = new URL(req.url);
  const slug = url.pathname.slice(1);
  
  if (!slug) {
    return new Response('Not Found', { status: 404 });
  }
  
  try {
    // 查找短链接
    const link = await prisma.link.findUnique({ where: { slug } });
    
    if (!link) {
      return new Response('Not Found', { status: 404 });
    }
    
    // 记录点击
    await prisma.click.create({
      data: {
        linkId: link.id,
        ipAddress: req.headers.get('x-forwarded-for') || req.headers.get('remote-addr'),
        userAgent: req.headers.get('user-agent'),
        referrer: req.headers.get('referer'),
      },
    });
    
    // 重定向到原始URL
    return new Response(null, {
      status: 302,
      headers: { Location: link.originalUrl },
    });
  } catch (error) {
    console.error('Redirect error:', error);
    return new Response('Internal Server Error', { status: 500 });
  }
};
```

```typescript
import { prisma } from '../db';

const authOptions = {
```

## 四、平台兼容性配置

### 1. Netlify (`netlify.toml`)
```toml
[build]
  command = "npm run build"
  publish = "dist"

[[redirects]]
  from = "/*"
  to = "/.netlify/functions/redirect"
  status = 200
  conditions = { Query = { NOT = ["path=/.netlify/*"] } }

[functions]
  node_bundler = "esbuild"
  external_node_modules = ["@prisma/client"]
```

### 2. Vercel (`vercel.json`)
```json
{
  "buildCommand": "npm run build",
  "outputDirectory": "dist",
  "rewrites": [
    {
      "source": "/api/:path*",
      "destination": "/functions/api/:path*"
    },
    {
      "source": "/(?!api/)(.*)",
      "destination": "/functions/redirect"

  ]
}
```
